5) Adapter (ربط منصتك)

> يفترض وجود endpoint يعيد JSON. إن اختلفت البنية، عدّل الـ parsing فقط.



# app/adapters/external_platform.py
import httpx, time
from app.config import EXTERNAL_API, REQUEST_TIMEOUT
from app.data.models import PlatformData

async def fetch_platform() -> PlatformData:
    async with httpx.AsyncClient(timeout=REQUEST_TIMEOUT) as client:
        r = await client.get(EXTERNAL_API)
        r.raise_for_status()
        data = r.json()

    # ✱ عدّل المفاتيح حسب شكل API لديك
    price = float(data.get("price", data.get("spot", 0)))
    signal = data.get("signal", data.get("live_signal", "WAIT"))
    confidence = float(data.get("confidence", data.get("score", 0)))
    zones = data.get("zones", data.get("liquidity_zones", {}))
    sentiment = data.get("sentiment", data.get("market_intent", None))

    return PlatformData(
        price=price,
        signal=signal,
        confidence=confidence,
        zones=zones,
        sentiment=sentiment,
        timestamp=time.time()
    )

6) Validation

# app/core/validator.py
import time
from app.config import STALE_THRESHOLD_SEC

def validate(p):
    penalty = 0.0
    flags = []

    if (time.time() - p.timestamp) > STALE_THRESHOLD_SEC:
        penalty += 0.2
        flags.append("stale_data")

    if p.price <= 0:
        penalty += 0.5
        flags.append("invalid_price")

    return penalty, flags

7) Agents

OrderFlow (بدون مزود عمق حقيقي → proxy)

# app/agents/orderflow.py
def run_orderflow(p):
    # Proxy بسيط يعتمد على الزخم/الإشارة
    delta = 1 if p.signal == "BUY" else -1 if p.signal == "SELL" else 0
    imbalance = "buyers" if delta > 0 else "sellers" if delta < 0 else "neutral"
    absorption = False  # يتطلب بيانات عمق حقيقية

    return {
        "delta": delta,
        "imbalance": imbalance,
        "absorption": absorption,
        "score": 0.6 if imbalance != "neutral" else 0.5
    }

Trap Engine

# app/agents/trap.py
def run_trap(p):
    trap_score = 0.0
    reasons = []

    # مثال: ثقة عالية + غياب محفزات واضحة => خطر فخ
    if p.confidence > 0.9 and (p.sentiment in [None, "neutral"]):
        trap_score = 0.4
        reasons.append("high_conf_no_catalyst")

    # قرب السعر من حدود سيولة عليا/دنيا (إن توفرت)
    # zones = {"R1":..., "S1":...}
    if p.zones:
        # منطق تقريبي
        if "R1" in p.zones and abs(p.price - float(p.zones["R1"])) < 5:
            trap_score += 0.2
            reasons.append("near_resistance")

    return {
        "trap_score": min(trap_score, 1.0),
        "reasons": reasons
    }

8) Fusion (دمج القرار)

# app/core/fusion.py
def fuse(p, orderflow, trap, macro, penalty):
    w1, w2, w3, w4 = 0.4, 0.25, 0.2, 0.15

    base = p.confidence
    of = orderflow["score"]
    ma = macro["macro_score"]
    tr = trap["trap_score"]

    confidence = (w1*base + w2*of + w3*ma - w4*tr) - penalty
    confidence = max(0.0, min(1.0, confidence))

    if confidence > 0.7:
        action = "BUY" if p.signal == "BUY" else "SELL" if p.signal == "SELL" else "WAIT"
    elif confidence < 0.4:
        action = "WAIT"
    else:
        action = "WAIT"

    reason = [
        f"base:{base:.2f}",
        f"orderflow:{of:.2f}",
        f"macro:{ma:.2f}",
        f"trap:{tr:.2f}",
        f"penalty:{penalty:.2f}"
    ]

    return action, confidence, reason

9) Risk Manager

# app/core/risk.py
from app.config import BASE_RISK, DAILY_DRAWDOWN_LIMIT

_state = {
    "drawdown_today": 0.0
}

def compute_risk(confidence, trap_score):
    if _state["drawdown_today"] > DAILY_DRAWDOWN_LIMIT:
        return {"allow": False, "position_size": 0.0, "reason": "daily_limit"}

    risk = BASE_RISK * confidence * (1 - trap_score)
    risk = max(0.0, min(risk, BASE_RISK))

    return {"allow": True, "position_size": risk}

# app/core/execution.py
def execute(action, risk, price):
    if not risk["allow"] or action == "WAIT":
        return {"status": "NO_TRADE"}

    # محاكاة
    return {
        "status": "EXECUTED",
        "action": action,
        "entry": price,
        "size": risk["position_size"]
    }

11) الخدمة (API + Pipeline)

# app/main.py
from fastapi import FastAPI
from app.adapters.external_platform import fetch_platform
from app.core.validator import validate
from app.agents.orderflow import run_orderflow
from app.agents.trap import run_trap
from app.agents.macro import run_macro
from app.core.fusion import fuse
from app.core.risk import compute_risk
from app.core.execution import execute

app = FastAPI()

@app.get("/decision")
async def decision():
    p = await fetch_platform()

    penalty, flags = validate(p)

    of = run_orderflow(p)
    tr = run_trap(p)
    ma = run_macro(p)

    action, conf, reason = fuse(p, of, tr, ma, penalty)

    risk = compute_risk(conf, tr["trap_score"])
    exec_result = execute(action, risk, p.price)

    return {
        "price": p.price,
        "platform_signal": p.signal,
        "final_action": action,
        "confidence": conf,
        "reason": reason + flags,
        "agents": {
            "orderflow": of,
            "trap": tr,
            "macro": ma
        },
        "risk": risk,
        "execution": exec_result
    }
