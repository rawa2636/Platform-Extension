import httpx, re

GITHUB_RAW = "https://raw.githubusercontent.com/zebbern/no-cost-ai/main/README.md"

async def fetch_repo():
    async with httpx.AsyncClient() as c:
        r = await c.get(GITHUB_RAW, timeout=10)
        return r.text

def extract_links(md):
    urls = re.findall(r'https?://[^\s)]+', md)
    return list(set(urls))

Health Check + Benchmark

import time, httpx

async def health_check(url):
    t0 = time.time()
    try:
        async with httpx.AsyncClient(timeout=5) as c:
            r = await c.get(url)
        return {"ok": r.status_code==200, "latency": (time.time()-t0)*1000}
    except:
        return {"ok": False, "latency": 9999}

async def benchmark(url):
    # طلب بسيط موحد
    payload = {"prompt": "ping", "max_tokens": 8}
    try:
        t0 = time.time()
        async with httpx.AsyncClient(timeout=8) as c:
            r = await c.post(url, json=payload)
        dt = (time.time()-t0)
        tokens = 8
        return {"tps": tokens/max(dt,1e-3), "success": r.status_code==200, "quality": 0.7}
    except:
        return {"tps": 0, "success": False, "quality": 0}

Rank + Store

def compute_score(x):
    if not x["ok"]:
        return 0
    lat = x["latency"]
    tps = x.get("tps",0)
    succ = 1.0 if x.get("success") else 0.0
    q = x.get("quality",0)

    return 0.35*(1/max(lat,1)) + 0.30*tps + 0.20*succ + 0.15*q

def rank_all(items):
    for it in items:
        it["score"] = compute_score(it)
    return sorted(items, key=lambda x: x["score"], reverse=True)
